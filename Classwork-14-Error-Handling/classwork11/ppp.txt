START

// 1. Read the configuration file safely
Define Dictionary config
Open file "config.txt" in read mode

For each line in the file:
    If the line contains "=":
        Split the line by "=" into 'key' and 'value'
        Remove whitespace
        config[key] = Convert to Float(value)

Close file "config.txt"

// Display the read configuration on the screen
For each (key, value) in config:
    Print "key=value"

// 2. Extract dimensions and convert them to integers
width = Convert to Integer(config["ancho"])
height = Convert to Integer(config["alto"])
max_iter = Convert to Integer(config["max_iter"])

// 3. Generate and save the fractal data
Open file "clase.csv" in write mode
Write the line "fila,columna,iteraciones" to the file

// Loop through each pixel of the grid
For row from 0 to (height - 1):
    For column from 0 to (width - 1):
        
        // Map the pixel coordinates to the complex plane
        real = config["real_min"] + (column / width) * (config["real_max"] - config["real_min"])
        imag = config["imag_min"] + (row / height) * (config["imag_max"] - config["imag_min"])
        
        c = Complex Number(real, imag)
        z = Complex Number(0, 0)
        iterations = 0
        
        // Calculate the fractal behavior
        While (Absolute Value(z) <= 2) AND (iterations < max_iter):
            z = (z * z) + c
            iterations = iterations + 1
            
        // Save the result in the CSV file
        Write the line "row,column,iterations" to the file "clase.csv"

Close file "clase.csv"

Print "DONE"

END
