PPP - Spanish Verb Conjugator
# Pseudocode written in plain English following class rules
# ← is used for assignment, # is used for comments, no Python syntax is used

# ---------------------------------------------
# DATA SETUP
# ---------------------------------------------
START

regularEndingsAR ← list of endings for -ar verbs for each person
regularEndingsER ← list of endings for -er verbs for each person
regularEndingsIR ← list of endings for -ir verbs for each person
personList ← list of the six persons (yo, tu, el/ella, nosotros, vosotros, ellos)

irregularVerbs ← table of fully conjugated irregular verbs
    # each entry has: infinitive, and the six conjugated forms
    # example entries: ser, estar, ir, tener, hacer, querer, poder, decir

score ← 0
totalQuestions ← 0

# ---------------------------------------------
# MAIN PROGRAM LOOP
# ---------------------------------------------
REPEAT
    DISPLAY "1. Conjugate a verb for one person"
    DISPLAY "2. Show full conjugation table for a verb"
    DISPLAY "3. Practice quiz mode"
    DISPLAY "4. Exit program"
    GET userChoice FROM user                         # INPUT

    IF userChoice = 1 THEN
        CALL ModeSingleConjugation
    ELSE IF userChoice = 2 THEN
        CALL ModeFullTable
    ELSE IF userChoice = 3 THEN
        CALL ModeQuiz
    ELSE IF userChoice = 4 THEN
        DISPLAY "Goodbye! / Adios!"
    ELSE
        DISPLAY "Invalid option, please choose 1 to 4"
    END IF

UNTIL userChoice = 4

END

# ---------------------------------------------
# MODE 1: SINGLE CONJUGATION
# ---------------------------------------------
PROCEDURE ModeSingleConjugation
    GET verb FROM user                                # INPUT
    DISPLAY personList WITH numbers
    GET personChoice FROM user                        # INPUT

    person ← personList[personChoice]
    result ← CALL GetConjugation(verb, person)         # PROCESS

    IF result = "INVALID" THEN
        DISPLAY "That verb is not -ar, -er, -ir, or not in the irregular list"
    ELSE
        DISPLAY verb + " for " + person + " is: " + result   # OUTPUT
    END IF
END PROCEDURE

# ---------------------------------------------
# MODE 2: FULL CONJUGATION TABLE (uses iteration)
# ---------------------------------------------
PROCEDURE ModeFullTable
    GET verb FROM user                                # INPUT
    DISPLAY "Conjugation table for " + verb

    FOR EACH person IN personList                      # ITERATION
        result ← CALL GetConjugation(verb, person)      # PROCESS
        IF result = "INVALID" THEN
            DISPLAY "That verb is not -ar, -er, -ir, or not in the irregular list"
            EXIT FOR
        ELSE
            DISPLAY person + " -> " + result            # OUTPUT
        END IF
    END FOR EACH
END PROCEDURE

# ---------------------------------------------
# MODE 3: QUIZ / PRACTICE MODE (uses iteration)
# ---------------------------------------------
PROCEDURE ModeQuiz
    GET numberOfQuestions FROM user                    # INPUT
    score ← 0

    FOR counter ← 1 TO numberOfQuestions                # ITERATION
        verb ← CALL PickRandomVerb()
        person ← CALL PickRandomPerson()
        correctAnswer ← CALL GetConjugation(verb, person)  # PROCESS

        DISPLAY "Conjugate: " + verb + " for " + person
        GET userAnswer FROM user                         # INPUT

        IF userAnswer = correctAnswer THEN
            DISPLAY "Correct!"                           # OUTPUT
            score ← score + 1
        ELSE
            DISPLAY "Incorrect. The right answer is: " + correctAnswer  # OUTPUT
        END IF
    END FOR

    DISPLAY "You scored " + score + " out of " + numberOfQuestions    # OUTPUT
END PROCEDURE

# ---------------------------------------------
# FUNCTION: GET CONJUGATION (uses decision)
# ---------------------------------------------
FUNCTION GetConjugation(verb, person)
    IF verb IS IN irregularVerbs THEN
        conjugatedForm ← look up irregularVerbs[verb][person]   # DECISION
        RETURN conjugatedForm
    ELSE
        ending ← last two letters of verb

        IF ending = "ar" THEN
            stem ← verb WITHOUT last two letters
            conjugatedForm ← stem + regularEndingsAR[person]    # DECISION
        ELSE IF ending = "er" THEN
            stem ← verb WITHOUT last two letters
            conjugatedForm ← stem + regularEndingsER[person]    # DECISION
        ELSE IF ending = "ir" THEN
            stem ← verb WITHOUT last two letters
            conjugatedForm ← stem + regularEndingsIR[person]    # DECISION
        ELSE
            RETURN "INVALID"
        END IF

        RETURN conjugatedForm
    END IF
END FUNCTION
